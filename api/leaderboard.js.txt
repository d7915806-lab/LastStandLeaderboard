export default async function handler(req, res) {

    if (req.method !== "POST") {
        return res.status(405).json({
            error: "POST only"
        });
    }

    const auth = req.headers.authorization;

    if (auth !== `Bearer ${process.env.LEADERBOARD_SECRET}`) {
        return res.status(401).json({
            error: "Unauthorized"
        });
    }

    const webhook = process.env.DISCORD_WEBHOOK_URL;

    if (!webhook) {
        return res.status(500).json({
            error: "Discord webhook is not configured"
        });
    }

    const data = req.body;

    function formatLeaderboard(title, players) {

        if (!players || players.length === 0) {
            return `**${title}**\nNo players yet.`;
        }

        return [
            `**${title}**`,
            ...players.map((player, index) => {
                return `${index + 1}. **${player.username}** — ${player.value.toLocaleString()}`;
            })
        ].join("\n");
    }

    const message = {
        username: "Last Stand: Waves",

        embeds: [
            {
                title: "🏆 LAST STAND: WAVES — ALL-TIME LEADERBOARD",
                description:
                    formatLeaderboard("💰 Top Money", data.money) +
                    "\n\n" +
                    formatLeaderboard("🌊 Highest Wave", data.wave) +
                    "\n\n" +
                    formatLeaderboard("♻️ Most Rebirths", data.rebirth),

                footer: {
                    text: "Last Stand: Waves • All-Time Records"
                }
            }
        ]
    };

    try {

        const response = await fetch(webhook, {
            method: "POST",

            headers: {
                "Content-Type": "application/json"
            },

            body: JSON.stringify(message)
        });

        if (!response.ok) {
            const text = await response.text();

            return res.status(500).json({
                error: "Discord rejected the message",
                details: text
            });
        }

        return res.status(200).json({
            success: true
        });

    } catch (error) {

        return res.status(500).json({
            error: error.message
        });
    }
}
